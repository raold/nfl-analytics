# NFL Analytics Production Cron Jobs
# ====================================
#
# Automated weekly workflow for predictions, odds tracking, and outcome collection
#
# Schedule:
#   - Tuesday 12pm ET: Generate QB passing predictions for upcoming week
#   - 3x daily (10am, 4pm, 10pm ET): Fetch odds from The Odds API
#   - Wednesday 6am ET: Backfill outcomes from completed games
#
# Setup:
#   1. Set ODDS_API_KEY environment variable
#   2. Run: crontab config/nfl_analytics_crontab.txt
#   3. Verify: crontab -l
#
# Logs:
#   - Predictions: /tmp/weekly_predictions.log
#   - Odds: /tmp/odds_fetcher.log
#   - Outcomes: /tmp/backfill_outcomes.log
#
# Free Tier Budget: 84 requests/month (500 limit - plenty of headroom)

# Environment variables (REQUIRED)
ODDS_API_KEY=your_api_key_here
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin

# Project directory (UPDATE THIS PATH)
NFL_ANALYTICS_DIR=/Users/dro/rice/nfl-analytics

# ====================================
# Weekly Prediction Generation
# ====================================
# Tuesday 12pm ET (17:00 UTC) - After Monday Night Football completes
# Auto-detects next upcoming week
0 17 * * 2 cd $NFL_ANALYTICS_DIR && uv run python scripts/generate_weekly_predictions.py >> /tmp/weekly_predictions_cron.log 2>&1

# ====================================
# Odds Tracking (3x daily)
# ====================================
# 10am ET (15:00 UTC) - Morning line
# Using smart odds script with priority-based fetching and quota tracking
0 15 * * * cd $NFL_ANALYTICS_DIR && uv run python py/ingest_odds_smart.py --priority high --max-calls 10 >> /tmp/odds_fetcher_cron.log 2>&1

# 4pm ET (21:00 UTC) - Afternoon movement
0 21 * * * cd $NFL_ANALYTICS_DIR && uv run python py/ingest_odds_smart.py --priority high --max-calls 10 >> /tmp/odds_fetcher_cron.log 2>&1

# 10pm ET (03:00 UTC next day) - Closing lines
0 3 * * * cd $NFL_ANALYTICS_DIR && uv run python py/ingest_odds_smart.py --priority urgent --max-calls 15 >> /tmp/odds_fetcher_cron.log 2>&1

# ====================================
# Outcome Backfill
# ====================================
# Wednesday 6am ET (11:00 UTC) - After all games complete (including MNF)
# Auto-detects all pending predictions
0 11 * * 3 cd $NFL_ANALYTICS_DIR && uv run python scripts/backfill_outcomes.py >> /tmp/backfill_outcomes_cron.log 2>&1

# ====================================
# Database Maintenance
# ====================================
# Daily 5am ET (10:00 UTC) - Refresh materialized views
# Runs after any overnight data updates
0 10 * * * cd $NFL_ANALYTICS_DIR && uv run python scripts/refresh_marts.py --all >> /tmp/refresh_marts_cron.log 2>&1

# ====================================
# Feature Staleness Monitoring
# ====================================
# Daily 6am ET (11:00 UTC) - Check feature freshness
# Runs after materialized view refresh to validate data currency
0 11 * * * cd $NFL_ANALYTICS_DIR && uv run python scripts/feature_staleness_monitor.py >> /tmp/feature_staleness_cron.log 2>&1
